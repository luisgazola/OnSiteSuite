cmake_minimum_required(VERSION 3.20)
project(OnSiteProbe VERSION 2.0.0 LANGUAGES CXX)
add_executable(onsite-probe onsite_probe.cpp)
target_compile_features(onsite-probe PRIVATE cxx_std_20)
if (MSVC)
 target_compile_options(onsite-probe PRIVATE /W4 /permissive-)
else()
 target_compile_options(onsite-probe PRIVATE -Wall -Wextra -Wpedantic -Wconversion)
endif()
